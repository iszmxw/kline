业务表：
# 邮箱路由节点配置表 `t_system_email_routing_node` 
CREATE TABLE `t_system_email_routing_node` (
  `id` bigint(20) NOT NULL AUTO_INCREMENT COMMENT '邮箱路由节点配置表ID',
  `email` varchar(64) DEFAULT '' COMMENT '邮箱账号',
  `channel` varchar(64) DEFAULT '' COMMENT '三方渠道',
  `channel_password` varchar(64) DEFAULT '' COMMENT '渠道密码',
  `channel_port` varchar(64) DEFAULT '' COMMENT '渠道端口号',
  `available_number` int(6) DEFAULT '300' COMMENT '剩余可用条数',
  `max_number` int(6) unsigned DEFAULT '300' COMMENT '每日最多使用条数',
  `sort` int(5) unsigned DEFAULT '1' COMMENT '优先排序',
  `valid` int(3) unsigned DEFAULT '1' COMMENT '有效状态 1有效 0失效',
  `creat_date` datetime DEFAULT NULL COMMENT '创建时间',
  `update_date` datetime DEFAULT NULL COMMENT '修改时间',
  `creator` varchar(64) DEFAULT '' COMMENT '创建人',
  `updater` varchar(64) DEFAULT '' COMMENT '修改人',
  `remarks` text COMMENT '备注',
  `nation` varchar(128) DEFAULT '' COMMENT '绑定国家',
  `item_code` varchar(64) DEFAULT '' COMMENT '应用编码',
  PRIMARY KEY (`id`) USING BTREE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

# 邮箱发送记录表 `t_system_email_send_log` 
CREATE TABLE `t_system_email_send_log` (
  `id` BIGINT(20) NOT NULL AUTO_INCREMENT COMMENT '邮箱发送记录表ID',
  `sending_mailbox` VARCHAR(64) DEFAULT '' COMMENT '发件邮箱账号',
  `send_total` INT(6) DEFAULT '' COMMENT '发送数量',
  `available_number` INT(6) DEFAULT '300' COMMENT '剩余可用条数',
  `creat_date` DATETIME DEFAULT NULL COMMENT '创建时间',
  `update_date` DATETIME DEFAULT NULL COMMENT '修改时间',
  `creator` VARCHAR(64) DEFAULT '' COMMENT '创建人',
  `updater` VARCHAR(64) DEFAULT '' COMMENT '修改人',
  `remarks` VARCHAR(255) DEFAULT '' COMMENT '备注',
  PRIMARY KEY (`id`) USING BTREE
) ENGINE=INNODB AUTO_INCREMENT=1 DEFAULT CHARSET=utf8mb4 ;

#三方邮箱默认配置表 `t_system_email_partner_config_set` 
CREATE TABLE `t_system_email_partner_config_set` (
  `id` BIGINT(20) NOT NULL AUTO_INCREMENT COMMENT '三方邮箱默认配置表ID',
  `channel` VARCHAR(64) DEFAULT '' COMMENT '三方渠道',
  `channel_port` VARCHAR(64) DEFAULT '' COMMENT '渠道端口号',
  `max_number` INT(6) UNSIGNED DEFAULT '300' COMMENT '每日最多使用条数',
  `valid` INT(3) UNSIGNED DEFAULT '1' COMMENT '有效状态 1有效 0失效',
  `creat_date` DATETIME DEFAULT NULL COMMENT '创建时间',
  `update_date` DATETIME DEFAULT NULL COMMENT '修改时间',
  `creator` VARCHAR(64) DEFAULT '' COMMENT '创建人',
  `updater` VARCHAR(64) DEFAULT '' COMMENT '修改人',
  `remarks` VARCHAR(255) DEFAULT '' COMMENT '备注',
  `nation` VARCHAR(128) DEFAULT '' COMMENT '绑定国家',
  PRIMARY KEY (`id`) USING BTREE
) ENGINE=INNODB AUTO_INCREMENT=1 DEFAULT CHARSET=utf8mb4 ;
